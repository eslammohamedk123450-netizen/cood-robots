const PERSONAS = {
  soly:   {name:"سولي",  head:"#6fa8ff", accent:"#1d4ed8", cheek:"#9ec5ff", lashes:false, hair:"short"},
  soleen: {name:"سولين", head:"#ff9ccf", accent:"#c026d3", cheek:"#ff6fae", lashes:true,  hair:"bow"}
};
// brow tilt (deg-ish offset), eye open (0-1), mouth: [curve, openness]
const EMOTIONS = {
  neutral:   {label:"عادي",     brow:0,   open:1,   mouth:[0,0]},
  listening: {label:"بيسمع",    brow:4,  open:1.1, mouth:[0,0.15]},
  thinking:  {label:"بيفكر",    brow:-6,  open:.8,  mouth:[-0.2,0], look:[14,-10]},
  speaking:  {label:"بيتكلم",   brow:0,   open:1,   mouth:[0.2,0.6], talk:true},
  happy:     {label:"سعيد",     brow:5,  open:.7,  mouth:[1,0.3]},
  sad:       {label:"حزين",     brow:14,  open:.9,  mouth:[-1,0]},
  surprised: {label:"متفاجئ",   brow:12, open:1.3, mouth:[0,1]}
};
const $ = id => document.getElementById(id);
let persona = "soly", emotion = "neutral", talkT = 0;

function drawMouth(curve, open){
  const y=225, w=60, c=curve*28, o=open*34;
  if(open<0.05) return `M${200-w} ${y} Q200 ${y+c*2} ${200+w} ${y}`;
  return `M${200-w} ${y} Q200 ${y+c*2} ${200+w} ${y} Q200 ${y+o*2+c} ${200-w} ${y} Z`;
}
function drawBrows(t){
  $("browL").setAttribute("d",`M95 ${78+t*.3} L165 ${78-t}`);
  $("browR").setAttribute("d",`M305 ${78+t*.3} L235 ${78-t}`);
}
function drawHair(){
  const h=$("hair"), p=PERSONAS[persona];
  const col = persona==="soly" ? "#1e293b" : "#7c1d6f";
  h.innerHTML = p.hair==="short"
    ? `<path d="M60 90 Q70 10 200 14 Q330 10 340 90 Q300 50 200 52 Q100 50 60 90Z" fill="${col}"/>`
    : `<path d="M50 120 Q40 8 200 12 Q360 8 350 120 Q330 40 200 46 Q70 40 50 120Z" fill="${col}"/>
       <path d="M300 22 L350 0 L350 44 Z M300 22 L250 0 L250 44 Z" fill="#fde047"/><circle cx="300" cy="22" r="9" fill="#f59e0b"/>`;
  $("lashes").innerHTML = p.lashes
    ? `<path d="M96 100 L84 88"/><path d="M108 92 L102 76"/><path d="M304 100 L316 88"/><path d="M292 92 L298 76"/>` : "";
}
function render(blink=1, mouthOpen=null){
  const e=EMOTIONS[emotion], look=e.look||[0,0];
  const ry=40*e.open*blink;
  for(const s of ["L","R"]){
    const cx = s==="L"?130:270;
    $("eye"+s).setAttribute("ry", Math.max(ry,2));
    $("pup"+s).setAttribute("cx", cx+look[0]);
    $("pup"+s).setAttribute("cy", 130+look[1]);
    $("pup"+s).setAttribute("r", blink<.3?0:14);
  }
  drawBrows(e.brow);
  $("mouth").setAttribute("d", drawMouth(e.mouth[0], mouthOpen ?? e.mouth[1]));
}
function apply(){
  const p=PERSONAS[persona], r=document.documentElement.style;
  r.setProperty("--head",p.head); r.setProperty("--accent",p.accent); r.setProperty("--cheek",p.cheek);
  $("name").textContent=p.name; $("status").textContent=EMOTIONS[emotion].label;
  document.querySelectorAll("#personas button").forEach(b=>b.classList.toggle("on",b.dataset.p===persona));
  document.querySelectorAll("#emotions button").forEach(b=>b.classList.toggle("on",b.dataset.e===emotion));
  drawHair(); render();
}
function setPersona(p){ if(PERSONAS[p]){persona=p; apply();} }
function setEmotion(e){ if(EMOTIONS[e]){emotion=e; apply();} }
window.robot = {setPersona, setEmotion};   // hook for later voice/AI code

for(const [k,v] of Object.entries(EMOTIONS)){
  const b=document.createElement("button"); b.textContent=v.label; b.dataset.e=k;
  b.onclick=()=>setEmotion(k); $("emotions").appendChild(b);
}
document.querySelectorAll("#personas button").forEach(b=>b.onclick=()=>setPersona(b.dataset.p));

// auto blink + talking mouth
function blink(){ let t=0; const id=setInterval(()=>{ t+=1; render(t<4?1-t/4:(t-4)/4>=1?1:(t-4)/4); if(t>=8){clearInterval(id);render();} },30); }
setInterval(()=>{ if(Math.random()<.6) blink(); },2500);
setInterval(()=>{ if(EMOTIONS[emotion].talk) render(1, .15+Math.random()*.7); },120);

const q=new URLSearchParams(location.search);
if(q.get("p")) persona=PERSONAS[q.get("p")]?q.get("p"):persona;
if(q.get("e")) emotion=EMOTIONS[q.get("e")]?q.get("e"):emotion;
apply();
